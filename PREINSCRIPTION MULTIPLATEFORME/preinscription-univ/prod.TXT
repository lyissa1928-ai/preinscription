pm2 delete all 2>/dev/null; pm2 save
rm -rf /var/www/uniportail /home/admin/web/uniportail-digital.com/uniportail
certbot delete --cert-name uniportail-digital.com 2>/dev/null || true
rm -rf /var/www/letsencrypt
rm -f /etc/nginx/conf.d/uniportail*.conf /etc/nginx/sites-enabled/uniportail-digital.com /etc/nginx/sites-available/uniportail-digital.com 2>/dev/null
/usr/local/hestia/bin/v-delete-web-domain USER_HESTIA uniportail-digital.com 2>/dev/null || true
rm -rf /home/admin/web/uniportail-digital.com /home/admin/conf/web/uniportail-digital.com* 2>/dev/null
nginx -t && systemctl reload nginx

apt update && apt install -y curl git certbot python3-certbot-nginx
curl -fsSL https://deb.nodesource.com/setup_20.x | bash - && apt install -y nodejs
npm install -g pm2

mkdir -p /home/admin/web/uniportail-digital.com && cd /home/admin/web/uniportail-digital.com
git clone https://github.com/lyissa1928-ai/preinscription.git uniportail
cd uniportail && git checkout main && git pull
npm run install:all && npm run build
chown -R USER_HESTIA:USER_HESTIA /home/admin/web/uniportail-digital.com/uniportail

cd /home/admin/web/uniportail-digital.com/uniportail/backend
cp .env.example .env
JWT_HEX=$(openssl rand -hex 32); sed -i "s/^JWT_SECRET=.*/JWT_SECRET=${JWT_HEX}/" .env
grep -q '^NODE_ENV=' .env && sed -i 's/^NODE_ENV=.*/NODE_ENV=production/' .env || echo NODE_ENV=production >> .env
grep -q '^PORT=' .env || echo PORT=5000 >> .env
grep -q '^CORS_ORIGINS=' .env || echo 'CORS_ORIGINS=https://uniportail-digital.com,https://www.uniportail-digital.com' >> .env

cd /home/admin/web/uniportail-digital.com/uniportail/backend
pm2 start server.js --name uniportail-api && pm2 save && pm2 startup systemd

mkdir -p /var/www/letsencrypt/.well-known/acme-challenge
chown -R www-data:www-data /var/www/letsencrypt

cat > /etc/nginx/conf.d/uniportail-digital.com.conf <<'EOF'
server {
    listen 80;
    listen [::]:80;
    server_name uniportail-digital.com www.uniportail-digital.com;
    root /home/admin/web/uniportail-digital.com/uniportail/frontend/dist;
    index index.html;
    client_max_body_size 25M;
    location ^~ /.well-known/acme-challenge/ {
        root /var/www/letsencrypt;
        default_type "text/plain";
        try_files $uri =404;
    }
    location /api/ {
        proxy_pass http://127.0.0.1:5000;
        proxy_http_version 1.1;
        proxy_set_header Host $host;
        proxy_set_header X-Real-IP $remote_addr;
        proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
        proxy_set_header X-Forwarded-Proto $scheme;
    }
    location /uploads/ {
        proxy_pass http://127.0.0.1:5000;
        proxy_set_header Host $host;
        proxy_set_header X-Real-IP $remote_addr;
        proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
        proxy_set_header X-Forwarded-Proto $scheme;
    }
    location / {
        try_files $uri $uri/ /index.html;
    }
}
EOF

nginx -t && systemctl reload nginx

certbot certonly --webroot -w /var/www/letsencrypt -d uniportail-digital.com -d www.uniportail-digital.com --agree-tos -m EMAIL@ --non-interactive || ( systemctl stop nginx; certbot certonly --standalone -d uniportail-digital.com -d www.uniportail-digital.com --agree-tos -m EMAIL@ --non-interactive; systemctl start nginx )

cat > /etc/nginx/conf.d/uniportail-digital.com.conf <<'EOF'
server {
    listen 80;
    listen [::]:80;
    server_name uniportail-digital.com www.uniportail-digital.com;
    location ^~ /.well-known/acme-challenge/ {
        root /var/www/letsencrypt;
        default_type "text/plain";
        try_files $uri =404;
    }
    location / { return 301 https://$host$request_uri; }
}
server {
    listen 443 ssl http2;
    listen [::]:443 ssl http2;
    server_name uniportail-digital.com www.uniportail-digital.com;
    ssl_certificate /etc/letsencrypt/live/uniportail-digital.com/fullchain.pem;
    ssl_certificate_key /etc/letsencrypt/live/uniportail-digital.com/privkey.pem;
    ssl_protocols TLSv1.2 TLSv1.3;
    root /home/admin/web/uniportail-digital.com/uniportail/frontend/dist;
    index index.html;
    client_max_body_size 25M;
    location /api/ {
        proxy_pass http://127.0.0.1:5000;
        proxy_http_version 1.1;
        proxy_set_header Host $host;
        proxy_set_header X-Real-IP $remote_addr;
        proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
        proxy_set_header X-Forwarded-Proto $scheme;
    }
    location /uploads/ {
        proxy_pass http://127.0.0.1:5000;
        proxy_set_header Host $host;
        proxy_set_header X-Real-IP $remote_addr;
        proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
        proxy_set_header X-Forwarded-Proto $scheme;
    }
    location / {
        try_files $uri $uri/ /index.html;
    }
}
EOF

nginx -t && systemctl reload nginx
pm2 restart uniportail-api
curl -sI https://uniportail-digital.com/ | head -5
certbot renew --dry-run
